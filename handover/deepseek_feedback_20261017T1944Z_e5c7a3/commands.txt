# [3] CPU AdamW 数值等价验证（对比 torch.optim.AdamW，20 步）

  ~/miniconda3/envs/thinktool/bin/python - <<PYEOF
import torch, sys
sys.path.insert(0, "selfplay/src")
from cpu_adamw import CPUStateAdamW
torch.manual_seed(0)
shape = (64, 32); w0 = torch.randn(shape)
w_ref = w0.clone().requires_grad_(True); w_cpu = w0.clone().requires_grad_(True)
ref = torch.optim.AdamW([w_ref], lr=1e-3, betas=(0.9,0.999), eps=1e-8, weight_decay=1e-2)
cpu = CPUStateAdamW([w_cpu], lr=1e-3, betas=(0.9,0.999), eps=1e-8, weight_decay=1e-2)
maxdiff = 0.0
for i in range(20):
    torch.manual_seed(100+i); g = torch.randn(shape)
    w_ref.grad = g.clone(); w_cpu.grad = g.clone()
    ref.step(); cpu.step()
    maxdiff = max(maxdiff, (w_ref.detach()-w_cpu.detach()).abs().max().item())
print("maxdiff = %.3e" % maxdiff)
PYEOF'
# → 20 步后最大逐元素差 = 0.000e+00（bit-exact）；state 全在 cpu
